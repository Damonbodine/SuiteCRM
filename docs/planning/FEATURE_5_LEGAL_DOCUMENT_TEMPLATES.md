# Feature 5: AI-Powered Legal Document Template Library

**Target:** Small Law Firms (1-10 Attorneys)  
**Implementation Time:** 2-3 days  
**Risk Level:** ⭐ Very Low  
**Business Value:** 🏆 High Daily Use  

---

## 🎯 FEATURE OVERVIEW

Transform SuiteCRM's existing Documents module into an intelligent legal template library that generates customized legal documents using AI. This feature leverages the proven Documents infrastructure while adding AI-powered template generation for 20+ common legal document types.

### Key Benefits
- **Instant Document Generation**: Create contracts, letters, motions in seconds
- **AI Customization**: Templates adapt to case details, client information, and firm preferences  
- **Zero Security Risk**: Uses existing Documents module security framework
- **Professional Templates**: Pre-built templates for common legal documents
- **Time Savings**: Reduce document creation from hours to minutes

### Phase 2: AI Template Generation Service (Day 2)

#### Step 4: Create AI Template Generator
```php
// custom/include/LegalTemplateAI.php
<?php
if (!defined('sugarEntry') || !sugarEntry) {
    die('Not A Valid Entry Point');
}

class LegalTemplateAI {
    
    private $openai_api_key;
    private $templates;
    
    public function __construct() {
        $this->openai_api_key = $GLOBALS['sugar_config']['openai_api_key'] ?? '';
        $this->loadTemplateDefinitions();
    }
    
    /**
     * Generate legal document using AI based on case and client data
     */
    public function generateDocument($templateType, $caseId, $clientId = null) {
        if (empty($this->openai_api_key)) {
            throw new Exception('OpenAI API key not configured');
        }
        
        // Load case and client data
        $caseData = $this->loadCaseData($caseId);
        $clientData = $clientId ? $this->loadClientData($clientId) : $this->extractClientFromCase($caseData);
        
        // Get template definition
        $template = $this->templates[$templateType] ?? null;
        if (!$template) {
            throw new Exception("Template type '$templateType' not found");
        }
        
        // Generate document content using AI
        $documentContent = $this->callOpenAI($template, $caseData, $clientData);
        
        // Save as new document
        return $this->saveGeneratedDocument($templateType, $documentContent, $caseId, $clientData);
    }
    
    private function loadTemplateDefinitions() {
        $this->templates = [
            'contract_retainer' => [
                'name' => 'Retainer Agreement',
                'prompt' => 'Generate a professional retainer agreement for a law firm. Include standard clauses for scope of representation, fee structure, client responsibilities, and termination. Use the provided case and client information to customize the document.',
                'variables' => ['firm_name', 'attorney_name', 'client_name', 'case_description', 'hourly_rate', 'retainer_amount']
            ],
            'letter_demand' => [
                'name' => 'Demand Letter', 
                'prompt' => 'Create a formal demand letter for legal action. Include factual background, legal basis for claim, specific demand for resolution, and consequences of non-compliance. Maintain a professional but firm tone.',
                'variables' => ['client_name', 'opposing_party', 'incident_date', 'damages_amount', 'demand_amount', 'deadline_date']
            ],
            'motion_dismiss' => [
                'name' => 'Motion to Dismiss',
                'prompt' => 'Draft a motion to dismiss following standard legal format. Include statement of facts, legal arguments, and prayer for relief. Reference relevant case law and statutes where appropriate.',
                'variables' => ['case_name', 'case_number', 'court_name', 'plaintiff_name', 'defendant_name', 'legal_basis', 'filing_date']
            ],
            'settlement_agreement' => [
                'name' => 'Settlement Agreement',
                'prompt' => 'Create a comprehensive settlement agreement resolving all claims between parties. Include payment terms, release clauses, confidentiality provisions, and enforcement mechanisms.',
                'variables' => ['party1_name', 'party2_name', 'settlement_amount', 'payment_schedule', 'case_description', 'effective_date']
            ],
            'letter_client_update' => [
                'name' => 'Client Update Letter',
                'prompt' => 'Write a professional client update letter summarizing case progress, recent developments, next steps, and any required client action. Maintain clear, non-legal language for client understanding.',
                'variables' => ['client_name', 'case_status', 'recent_developments', 'next_steps', 'attorney_name', 'update_date']
            ]
        ];
    }
    
    private function callOpenAI($template, $caseData, $clientData) {
        $systemPrompt = "You are a legal document drafting AI assistant for a law firm. Generate professional, legally sound documents based on the provided template and case information. Include all necessary legal language while keeping the document clear and professionally formatted.";
        
        $userPrompt = $this->buildPrompt($template, $caseData, $clientData);
        
        $data = [
            'model' => 'gpt-4',
            'messages' => [
                [
                    'role' => 'system',
                    'content' => $systemPrompt
                ],
                [
                    'role' => 'user', 
                    'content' => $userPrompt
                ]
            ],
            'max_tokens' => 3000,
            'temperature' => 0.3
        ];
        
        $ch = curl_init();
        curl_setopt($ch, CURLOPT_URL, 'https://api.openai.com/v1/chat/completions');
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
        curl_setopt($ch, CURLOPT_POST, 1);
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
        curl_setopt($ch, CURLOPT_HTTPHEADER, [
            'Content-Type: application/json',
            'Authorization: Bearer ' . $this->openai_api_key
        ]);
        
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);
        
        if ($httpCode !== 200) {
            throw new Exception("OpenAI API error: HTTP $httpCode - $response");
        }
        
        $result = json_decode($response, true);
        
        if (!isset($result['choices'][0]['message']['content'])) {
            throw new Exception('Invalid OpenAI response format');
        }
        
        return $result['choices'][0]['message']['content'];
    }
    
    private function buildPrompt($template, $caseData, $clientData) {
        $prompt = $template['prompt'] . "\n\n";
        $prompt .= "CASE INFORMATION:\n";
        $prompt .= "Case Name: " . ($caseData['name'] ?? 'N/A') . "\n";
        $prompt .= "Case Type: " . ($caseData['type'] ?? 'N/A') . "\n"; 
        $prompt .= "Status: " . ($caseData['status'] ?? 'N/A') . "\n";
        $prompt .= "Description: " . ($caseData['description'] ?? 'N/A') . "\n\n";
        
        $prompt .= "CLIENT INFORMATION:\n";
        $prompt .= "Client Name: " . ($clientData['name'] ?? 'N/A') . "\n";
        $prompt .= "Email: " . ($clientData['email'] ?? 'N/A') . "\n";
        $prompt .= "Phone: " . ($clientData['phone'] ?? 'N/A') . "\n";
        $prompt .= "Address: " . ($clientData['address'] ?? 'N/A') . "\n\n";
        
        $prompt .= "LAW FIRM INFORMATION:\n";
        $prompt .= "Firm Name: [Your Law Firm Name]\n";
        $prompt .= "Attorney: " . ($GLOBALS['current_user']->full_name ?? 'N/A') . "\n";
        $prompt .= "Date: " . date('F j, Y') . "\n\n";
        
        $prompt .= "Please generate a complete, professional legal document following standard legal formatting conventions.";
        
        return $prompt;
    }
    
    private function loadCaseData($caseId) {
        $case = BeanFactory::retrieveBean('Cases', $caseId);
        if (!$case) {
            throw new Exception("Case not found: $caseId");
        }
        
        return [
            'id' => $case->id,
            'name' => $case->name,
            'type' => $case->type,
            'status' => $case->status,
            'priority' => $case->priority,
            'description' => $case->description,
            'date_entered' => $case->date_entered
        ];
    }
    
    private function loadClientData($clientId) {
        // Try Contacts first, then Accounts
        $contact = BeanFactory::retrieveBean('Contacts', $clientId);
        if ($contact) {
            return [
                'id' => $contact->id,
                'name' => $contact->full_name,
                'email' => $contact->email1,
                'phone' => $contact->phone_work,
                'address' => $contact->primary_address_street
            ];
        }
        
        $account = BeanFactory::retrieveBean('Accounts', $clientId);
        if ($account) {
            return [
                'id' => $account->id,
                'name' => $account->name,
                'email' => $account->email1,
                'phone' => $account->phone_office,
                'address' => $account->billing_address_street
            ];
        }
        
        throw new Exception("Client not found: $clientId");
    }
    
    private function extractClientFromCase($caseData) {
        // For now, return placeholder data
        // In full implementation, would extract from case relationships
        return [
            'name' => '[Client Name from Case]',
            'email' => '[Client Email]',
            'phone' => '[Client Phone]',
            'address' => '[Client Address]'
        ];
    }
    
    private function saveGeneratedDocument($templateType, $content, $caseId, $clientData) {
        $document = BeanFactory::newBean('Documents');
        
        // Set document properties
        $template = $this->templates[$templateType];
        $document->name = $template['name'] . ' - ' . $clientData['name'] . ' - ' . date('Y-m-d');
        $document->document_type_c = $templateType;
        $document->is_template_c = 0; // This is a generated document, not a template
        $document->ai_generated_c = 1;
        $document->assigned_user_id = $GLOBALS['current_user']->id;
        
        // Create file from content
        $filename = sanitize_file_name($document->name) . '.txt';
        $uploadDir = rtrim($GLOBALS['sugar_config']['upload_dir'], '/');
        $filePath = $uploadDir . '/' . $document->id . '_' . $filename;
        
        if (file_put_contents($filePath, $content) === false) {
            throw new Exception('Failed to save generated document file');
        }
        
        $document->filename = $filename;
        $document->file_mime_type = 'text/plain';
        
        $document->save();
        
        // Create relationship to case
        if ($caseId) {
            $case = BeanFactory::retrieveBean('Cases', $caseId);
            if ($case) {
                $case->load_relationship('documents');
                $case->documents->add($document->id);
            }
        }
        
        return $document;
    }
}
?>
```

#### Step 5: Document Generation Controller  
```php
// custom/modules/Documents/generateTemplate.php
<?php
if (!defined('sugarEntry') || !sugarEntry) {
    die('Not A Valid Entry Point');
}

require_once('custom/include/LegalTemplateAI.php');

global $current_user;

// Validate authentication
if (empty($current_user->id)) {
    sugar_die("Authentication required");
}

// Validate permissions  
if (!ACLController::checkAccess('Documents', 'edit', true)) {
    sugar_die("Insufficient permissions to generate documents");
}

try {
    // Get and validate parameters
    $templateType = $_POST['template_type'] ?? '';
    $caseId = $_POST['case_id'] ?? '';
    $clientId = $_POST['client_id'] ?? '';
    
    if (empty($templateType)) {
        throw new Exception('Template type is required');
    }
    
    if (empty($caseId)) {
        throw new Exception('Case ID is required');
    }
    
    // Generate document
    $templateAI = new LegalTemplateAI();
    $document = $templateAI->generateDocument($templateType, $caseId, $clientId);
    
    // Log the generation
    $GLOBALS['log']->info("AI document generated: {$document->name} by user: {$current_user->user_name}");
    
    // Redirect to generated document
    SugarApplication::redirect("index.php?module=Documents&action=DetailView&record={$document->id}");
    
} catch (Exception $e) {
    $GLOBALS['log']->error("Document generation failed: " . $e->getMessage());
    
    // Redirect back with error
    $errorMsg = urlencode("Document generation failed: " . $e->getMessage());
    SugarApplication::redirect("index.php?module=Cases&action=DetailView&record={$caseId}&error={$errorMsg}");
}
?>
```

function openAIDocumentModal(caseId) {
    document.getElementById('modalCaseId').value = caseId;
    $('#aiDocumentModal').modal('show');
    loadCaseClients(caseId);

function loadCaseClients(caseId) {
    // Load clients related to this case
    fetch(`index.php?module=Cases&action=getRelatedClients&record=${caseId}`)
        .then(response => response.json())
        .then(data => {
            const clientSelect = document.getElementById('client_id');
            clientSelect.innerHTML = '<option value="">Auto-detect from case</option>';
            
            data.clients.forEach(client => {
                const option = document.createElement('option');
                option.value = client.id;
                option.textContent = client.name;
                clientSelect.appendChild(option);
            });
        })
        .catch(error => {
            console.error('Error loading clients:', error);
        });
}
